   120→// ─── Hypothesis tree helpers ───
   121→interface TreeNode {
   122→  hyp: HypothesisEvent
   123→  children: TreeNode[]
   124→}
   125→
   126→function buildTree(hypotheses: HypothesisEvent[]): TreeNode[] {
   127→  const map = new Map<string, TreeNode>()
   128→  const roots: TreeNode[] = []
   129→  for (const h of hypotheses) {
   130→    map.set(h.hypId, { hyp: h, children: [] })
   131→  }
   132→  for (const h of hypotheses) {
   133→    const node = map.get(h.hypId)!
   134→    if (h.parentId && map.has(h.parentId)) {
   135→      map.get(h.parentId)!.children.push(node)
   136→    } else {
   137→      roots.push(node)
   138→    }
   139→  }
   140→  return roots
   141→}
   142→
   143→interface LayoutNode {
   144→  x: number
   145→  y: number
   146→  node: TreeNode
   147→}
   148→
   149→function layoutTree(roots: TreeNode[]): LayoutNode[] {
   150→  const nodeW = 180
   151→  const nodeH = 48
   152→  const hGap = 24
   153→  const vGap = 70
   154→
   155→  function subtreeWidth(n: TreeNode): number {
   156→    if (n.children.length === 0) return nodeW
   157→    const cw = n.children.reduce((s, c) => s + subtreeWidth(c) + hGap, -hGap)
   158→    return Math.max(nodeW, cw)
   159→  }
   160→
   161→  const result: LayoutNode[] = []
   162→
   163→  function lay(node: TreeNode, x: number, y: number) {
   164→    const sw = subtreeWidth(node)
   165→    const cx = x + sw / 2
   166→    result.push({ x: cx - nodeW / 2, y, node })
   167→    if (node.children.length > 0) {
   168→      const cw = node.children.reduce((s, c) => s + subtreeWidth(c) + hGap, -hGap)
   169→      let childX = cx - cw / 2
   170→      for (const child of node.children) {
   171→        const childW = subtreeWidth(child)
   172→        lay(child, childX, y + nodeH + vGap)
   173→        childX += childW + hGap
   174→      }
   175→    }
   176→  }
   177→
   178→  let x = 0
   179→  for (const root of roots) {
   180→    const w = subtreeWidth(root)
   181→    lay(root, x, 0)
   182→    x += w + hGap * 2
   183→  }
   184→  return result
   185→}

   209→// ─── Stat counter ───
   210→function StatCounter({ icon: Icon, label, value, unit, color, pulse }: {
   211→  icon: React.ElementType; label: string; value: string | number; unit?: string; color?: string; pulse?: boolean
   212→}) {
   213→  const [displayVal, setDisplayVal] = useState(0)
   214→  const prev = useRef(0)
   215→  useEffect(() => {
   216→    const t = typeof value === 'number' ? value : parseFloat(String(value)) || 0
   217→    const s = prev.current, d = t - s
   218→    if (d === 0) return
   219→    const start = Date.now()
   220→    const tick = () => {
   221→      const p = Math.min((Date.now() - start) / 400, 1)
   222→      setDisplayVal(s + d * (1 - Math.pow(1 - p, 3)))
   223→      if (p < 1) requestAnimationFrame(tick); else prev.current = t
   224→    }
   225→    requestAnimationFrame(tick)
   226→  }, [value])
   227→  const f = typeof value === 'string' ? value :
   228→    Number.isInteger(value) ? Math.round(displayVal).toLocaleString() : displayVal.toFixed(1)
   229→  return (
   230→    <div className="relative">
   231→      {pulse && (
   232→        <span className="absolute -inset-1 rounded-xl border-2 border-primary animate-pulse-ring pointer-events-none" />
   233→      )}
   234→      <div className="flex items-center gap-3 p-3 rounded-lg bg-secondary/40 border border-border/30 relative z-[1]">
   235→        <div className={`p-2 rounded-md ${color || 'bg-primary/10 text-primary'}`}><Icon className="w-4 h-4" /></div>
   236→        <div className="min-w-0">
   237→          <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{label}</p>
   238→          <p className={`font-terminal text-lg font-bold ${color || 'text-foreground'}`}>{f}{unit && <span className="text-xs text-muted-foreground ml-0.5">{unit}</span>}</p>
   239→        </div>
   240→      </div>
   241→    </div>
   242→  )
   243→}

   442→// ─── Query Timing Sparkline ───
   443→function QueryTimingChart({ data }: { data: Array<{ index: number; dbMs: number }> }) {
   444→  if (data.length < 2) return null
   445→  const maxDbMs = Math.max(...data.map(d => d.dbMs), 1)
   446→  const w = 200
   447→  const h = 48
   448→  const pad = 2
   449→  const points = data.map((d, i) => ({
   450→    x: pad + (i / (data.length - 1)) * (w - 2 * pad),
   451→    y: pad + (1 - d.dbMs / maxDbMs) * (h - 2 * pad),
   452→  }))
   453→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x.toFixed(1)} ${p.y.toFixed(1)}`).join(' ')
   454→  const areaPath = `${linePath} L ${(points[points.length - 1].x).toFixed(1)} ${h} L ${(points[0].x).toFixed(1)} ${h} Z`
   455→  return (
   456→    <div className="mt-3">
   457→      <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium mb-1.5">Query Timing</p>
   458→      <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-12">
   459→        <defs>
   460→          <linearGradient id="timingGrad" x1="0" y1="0" x2="0" y2="1">
   461→            <stop offset="0%" stopColor="oklch(0.78 0.17 72)" stopOpacity="0.3" />
   462→            <stop offset="100%" stopColor="oklch(0.78 0.17 72)" stopOpacity="0.05" />
   463→          </linearGradient>
   464→        </defs>
   465→        <path d={areaPath} fill="url(#timingGrad)" />
   466→        <path d={linePath} fill="none" stroke="oklch(0.78 0.17 72)" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   467→        {points.map((p, i) => (
   468→          <circle key={i} cx={p.x.toFixed(1)} cy={p.y.toFixed(1)} r="1.5" fill="oklch(0.78 0.17 72)" />
   469→        ))}

   475→// ─── Hypothesis Tree Visualization ───
   476→function HypothesisTreeView({ hypotheses }: { hypotheses: HypothesisEvent[] }) {
   477→  const roots = buildTree(hypotheses)
   478→  if (roots.length === 0) return (
   479→    <div className="text-center py-12 text-muted-foreground">
   480→      <Network className="w-10 h-10 mx-auto mb-3 opacity-20" />
   481→      <p className="text-sm">Hypothesis tree will appear as the investigation progresses.</p>
   482→    </div>
   483→  )
   484→  const nodes = layoutTree(roots)
   485→  const nodeW = 180
   486→  const nodeH = 48
   487→  const maxX = Math.max(...nodes.map(n => n.x + nodeW)) + 30
   488→  const maxY = Math.max(...nodes.map(n => n.y + nodeH)) + 30
   489→  const nodeMap = new Map(nodes.map(n => [n.node.hyp.hypId, n]))
   490→  const edges: Array<{ x1: number; y1: number; x2: number; y2: number }> = []
   491→  for (const n of nodes) {
   492→    for (const child of n.node.children) {
   493→      const cn = nodeMap.get(child.hyp.hypId)
   494→      if (cn) {
   495→        edges.push({ x1: n.x + nodeW / 2, y1: n.y + nodeH, x2: cn.x + nodeW / 2, y2: cn.y })
   496→      }
   497→    }
   498→  }
   499→  const getVerdictFill = (v?: string) => {
   500→    if (v === 'CONFIRMED') return 'oklch(0.7 0.15 160 / 0.12)'
   501→    if (v === 'REFUTED') return 'oklch(0.65 0.25 25 / 0.12)'
   502→    if (v === 'INCONCLUSIVE') return 'oklch(0.75 0.15 85 / 0.12)'
   503→    return 'oklch(0.2 0 270 / 0.3)'
   504→  }
   505→  const getVerdictStroke = (v?: string) => {
   506→    if (v === 'CONFIRMED') return 'oklch(0.7 0.15 160)'
   507→    if (v === 'REFUTED') return 'oklch(0.65 0.25 25)'
   508→    if (v === 'INCONCLUSIVE') return 'oklch(0.75 0.15 85)'
   509→    return 'oklch(0.4 0 270)'
   510→  }
   511→  return (
   512→    <div className="overflow-x-auto">
   513→      <svg viewBox={`0 0 ${maxX} ${maxY}`} className="w-full min-w-[500px]" style={{ minHeight: '250px' }}>
   514→        {edges.map((e, i) => (
   515→          <line key={i} x1={e.x1} y1={e.y1} x2={e.x2} y2={e.y2} stroke="oklch(0.3 0 270)" strokeWidth="1.5" />
   516→        ))}
   517→        {nodes.map(n => {
   518→          const v = n.node.hyp.verdict
   519→          const fill = getVerdictFill(v)
   520→          const stroke = getVerdictStroke(v)
   521→          const label = n.node.hyp.statement.length > 28 ? n.node.hyp.statement.substring(0, 28) + '...' : n.node.hyp.statement
   522→          const subText = v ? `${v} ${n.node.hyp.confidence !== undefined ? `${Math.round(n.node.hyp.confidence * 100)}%` : ''}` : 'PENDING'
   523→          return (
   524→            <g key={n.node.hyp.hypId}>
   525→              <rect x={n.x} y={n.y} width={nodeW} height={nodeH} rx={8} fill={fill} stroke={stroke} strokeWidth="1.5" />
   526→              <text x={n.x + nodeW / 2} y={n.y + 19} textAnchor="middle" fill="oklch(0.9 0 0)" fontSize="10" fontWeight="500">{label}</text>
   527→              <text x={n.x + nodeW / 2} y={n.y + 36} textAnchor="middle" fill={stroke} fontSize="10" fontWeight="600">{subText}</text>
   528→            </g>
   529→          )
   530→        })}
   531→      </svg>
   532→    </div>
   533→  )
   534→}
