    97→// ─── Stat counter ───
    98→function StatCounter({ icon: Icon, label, value, unit, color }: {
    99→  icon: React.ElementType; label: string; value: string | number; unit?: string; color?: string
   100→}) {
   101→  const [displayVal, setDisplayVal] = useState(0)
   102→  const prev = useRef(0)
   103→  useEffect(() => {
   104→    const t = typeof value === 'number' ? value : parseFloat(String(value)) || 0
   105→    const s = prev.current, d = t - s
   106→    if (d === 0) return
   107→    const start = Date.now()
   108→    const tick = () => {
   109→      const p = Math.min((Date.now() - start) / 400, 1)
   110→      setDisplayVal(s + d * (1 - Math.pow(1 - p, 3)))
   111→      if (p < 1) requestAnimationFrame(tick); else prev.current = t
   112→    }
   113→    requestAnimationFrame(tick)
   114→  }, [value])
   115→  const f = typeof value === 'string' ? value :
   116→    Number.isInteger(value) ? Math.round(displayVal).toLocaleString() : displayVal.toFixed(1)
   117→  return (
   118→    <div className="flex items-center gap-3 p-3 rounded-lg bg-secondary/40 border border-border/30">
   119→      <div className={`p-2 rounded-md ${color || 'bg-primary/10 text-primary'}`}><Icon className="w-4 h-4" /></div>
   120→      <div className="min-w-0">
   121→        <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{label}</p>
   122→        <p className={`font-terminal text-lg font-bold ${color || 'text-foreground'}`}>{f}{unit && <span className="text-xs text-muted-foreground ml-0.5">{unit}</span>}</p>
   123→      </div>
   124→    </div>
   125→  )
   126→}

   239→export default function HomePage() {

   242→  const [socketConnected, setSocketConnected] = useState(false)
   243→  const [mobileMenu, setMobileMenu] = useState(false)
   244→  const [expandedHyps, setExpandedHyps] = useState<Set<string>>(new Set())
   245→  const [showExample, setShowExample] = useState(false)
   246→  const timelineRef = useRef<HTMLDivElement>(null)
   247→  const startTimeRef = useRef(0)
   248→  const elapsedRef = useRef<ReturnType<typeof setInterval> | null>(null)
   249→
   250→  const toggleHyp = useCallback((id: string) => {
   251→    setExpandedHyps(p => {
   252→      const n = new Set(p)
   253→      if (n.has(id)) { n.delete(id) } else { n.add(id) }
   254→      return n
   255→    })
   256→  }, [])
   257→
   258→  // Socket
   259→  useEffect(() => {
   260→    let cancelled = false, s: any = null
   261→    const init = async () => {
   262→      const { io } = await import('socket.io-client')
   263→      if (cancelled) return
   264→      s = io('/?XTransformPort=' + SERVICE_PORT, { path: '/', transports: ['polling', 'websocket'], forceNew: true, reconnection: true, reconnectionAttempts: 5, timeout: 30000 })
   265→      socketRef.current = s
   266→      s.on('connect', () => setSocketConnected(true))
   267→      s.on('connect_error', () => setSocketConnected(false))
   268→      s.on('disconnect', () => setSocketConnected(false))
   269→      s.on('investigation', (ev: any) => {
   270→        const { type, data } = ev
   271→        switch (type) {
   272→          case 'started': { store.setPhase('running'); store.setCaseFile(null as any); startTimeRef.current = Date.now(); store.addLog('system', `Investigation ${data.runId} started`); if (elapsedRef.current) clearInterval(elapsedRef.current); elapsedRef.current = setInterval(() => store.updateStats({ elapsedMs: Date.now() - startTimeRef.current }), 100); break }
   273→          case 'root_hypotheses': { store.addLog('hypothesis', `Generated ${data.hypotheses.length} root hypotheses`); for (const h of data.hypotheses) store.addHypothesis({ hypId: h.hypId, parentId: null, depth: 0, statement: h.statement, rationale: h.rationale, queryNum: 0, timestamp: ev.timestamp }); break }
   274→          case 'progress': store.setPhase('running'); break
   275→          case 'planned': store.updateHypothesis(data.hypId, { sql: data.sql, queryNum: data.queryNum }); store.addLog('sql', `Q${data.queryNum}: SQL planned`); break
   276→          case 'executed': store.updateHypothesis(data.hypId, { dbMs: data.dbMs, rowsReturned: data.rowsReturned, rowsScanned: data.rowsScanned }); store.updateStats({ queriesExecuted: data.totalQueries, rowsInScope: (store.stats.rowsInScope || 0) + (data.rowsScanned || 0), totalDbMs: data.totalDbMs }); break
   277→          case 'judged': store.updateHypothesis(data.hypId, { verdict: data.verdict, confidence: data.confidence, reasoning: data.reasoning }); if (data.verdict === 'CONFIRMED') store.updateStats({ hypothesesConfirmed: data.confirmed }); else if (data.verdict === 'REFUTED') store.updateStats({ hypothesesKilled: data.killed }); store.addLog('verdict', `Q${data.queryNum}: ${data.verdict} (${Math.round(data.confidence * 100)}%) — ${data.statement.substring(0, 60)}...`); break
