   127→// ─── Stat counter ───
   128→function StatCounter({ icon: Icon, label, value, unit, color }: {
   129→  icon: React.ElementType; label: string; value: string | number; unit?: string; color?: string
   130→}) {
   131→  const [displayVal, setDisplayVal] = useState(0)
   132→  const prev = useRef(0)
   133→  useEffect(() => {
   134→    const t = typeof value === 'number' ? value : parseFloat(String(value)) || 0
   135→    const s = prev.current, d = t - s
   136→    if (d === 0) return
   137→    const start = Date.now()
   138→    const tick = () => {
   139→      const p = Math.min((Date.now() - start) / 400, 1)
   140→      setDisplayVal(s + d * (1 - Math.pow(1 - p, 3)))
   141→      if (p < 1) requestAnimationFrame(tick); else prev.current = t
   142→    }
   143→    requestAnimationFrame(tick)
   144→  }, [value])
   145→  const f = typeof value === 'string' ? value :
   146→    Number.isInteger(value) ? Math.round(displayVal).toLocaleString() : displayVal.toFixed(1)
   147→  return (
   148→    <div className="flex items-center gap-3 p-3 rounded-lg bg-secondary/40 border border-border/30">
   149→      <div className={`p-2 rounded-md ${color || 'bg-primary/10 text-primary'}`}><Icon className="w-4 h-4" /></div>
   150→      <div className="min-w-0">
   151→        <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{label}</p>
   152→        <p className={`font-terminal text-lg font-bold ${color || 'text-foreground'}`}>{f}{unit && <span className="text-xs text-muted-foreground ml-0.5">{unit}</span>}</p>
   153→      </div>
   154→    </div>
   155→  )
   156→}
