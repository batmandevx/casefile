    47→// ─── Counter stat component ───
    48→function StatCounter({ icon: Icon, label, value, unit, color }: {
    49→  icon: React.ElementType; label: string; value: string | number; unit?: string; color?: string
    50→}) {
    51→  const [displayVal, setDisplayVal] = useState(0)
    52→  const prevVal = useRef(0)
    53→
    54→  useEffect(() => {
    55→    const target = typeof value === 'number' ? value : parseFloat(String(value)) || 0
    56→    const start = prevVal.current
    57→    const diff = target - start
    58→    if (diff === 0) return
    59→
    60→    const duration = 400
    61→    const startTime = Date.now()
    62→
    63→    const animate = () => {
    64→      const elapsed = Date.now() - startTime
    65→      const progress = Math.min(elapsed / duration, 1)
    66→      const eased = 1 - Math.pow(1 - progress, 3)
    67→      setDisplayVal(start + diff * eased)
    68→      if (progress < 1) requestAnimationFrame(animate)
    69→      else prevVal.current = target
    70→    }
    71→    requestAnimationFrame(animate)
    72→  }, [value])
    73→
    74→  const formatted = typeof value === 'string' ? value :
    75→    Number.isInteger(value) ? Math.round(displayVal).toLocaleString() :
    76→      displayVal.toFixed(1)
    77→
    78→  return (
    79→    <div className="flex items-center gap-3 p-3 rounded-lg bg-secondary/50">
    80→      <div className={`p-2 rounded-md ${color || 'bg-primary/10 text-primary'}`}>
    81→        <Icon className="w-4 h-4" />
    82→      </div>
    83→      <div className="min-w-0">
    84→        <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{label}</p>
    85→        <p className={`font-terminal text-lg font-bold ${color || 'text-foreground'}`}>
    86→          {formatted}{unit && <span className="text-xs text-muted-foreground ml-0.5">{unit}</span>}
    87→        </p>
    88→      </div>
    89→    </div>
    90→  )
    91→}

    94→function HypothesisItem({ hyp, isExpanded, onToggle }: {
    95→  hyp: HypothesisEvent; isExpanded: boolean; onToggle: () => void
    96→}) {
    97→  const verdictColor = VERDICT_COLORS[hyp.verdict || ''] || 'border-muted-foreground/30'
    98→  const verdictBg = VERDICT_BG[hyp.verdict || ''] || 'bg-muted/30'
    99→
   100→  return (
   101→    <motion.div
   102→      initial={{ opacity: 0, x: 20 }}
   103→      animate={{ opacity: 1, x: 0 }}
   104→      transition={{ duration: 0.3 }}
   105→      className={`border rounded-lg overflow-hidden ${verdictColor} ${verdictBg} animate-slide-in`}
   106→    >
   107→      <button
   108→        onClick={onToggle}
   109→        className="w-full text-left p-3 flex items-start gap-3 hover:bg-white/[0.02] transition-colors"
   110→      >
   111→        <div className="mt-0.5 shrink-0">
   112→          {VERDICT_ICONS[hyp.verdict || ''] || <HelpCircle className="w-4 h-4 text-muted-foreground" />}
   113→        </div>
   114→        <div className="flex-1 min-w-0">
   115→          <div className="flex items-center gap-2 flex-wrap">
   116→            <span className="font-terminal text-xs text-muted-foreground">Q{hyp.queryNum}</span>
   117→            {hyp.verdict && (
   118→              <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${verdictColor}`}>
   119→                {hyp.verdict}
   120→              </Badge>
   121→            )}
   122→            {hyp.confidence !== undefined && hyp.confidence > 0 && (
   123→              <span className="font-terminal text-xs text-muted-foreground">
   124→                {Math.round(hyp.confidence * 100)}%
   125→              </span>
   126→            )}
   127→          </div>
   128→          <p className="text-sm mt-1 leading-relaxed">{hyp.statement}</p>
   129→          {hyp.reasoning && !isExpanded && (
   130→            <p className="text-xs text-muted-foreground mt-1 line-clamp-1">{hyp.reasoning}</p>
   131→          )}
   132→        </div>
   133→        <ChevronDown className={`w-4 h-4 text-muted-foreground shrink-0 mt-1 transition-transform ${isExpanded ? 'rotate-180' : ''}`} />
   134→      </button>
   135→
   136→      <AnimatePresence>
   137→        {isExpanded && (
   138→          <motion.div
   139→            initial={{ height: 0, opacity: 0 }}
   140→            animate={{ height: 'auto', opacity: 1 }}
   141→            exit={{ height: 0, opacity: 0 }}
   142→            transition={{ duration: 0.2 }}
   143→            className="overflow-hidden"
   144→          >
   145→            <div className="px-3 pb-3 space-y-3">
   146→              {hyp.sql && (
   147→                <div>
   148→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground mb-1.5 font-medium">SQL</p>
   149→                  <div className="rounded-md overflow-hidden bg-black/30 max-h-48 overflow-y-auto">
   150→                    <SyntaxHighlighter
   151→                      language="sql"
   152→                      style={oneDark}
   153→                      customStyle={{ margin: 0, fontSize: '12px', background: 'transparent' }}
   154→                    >
   155→                      {hyp.sql}
   156→                    </SyntaxHighlighter>
   157→                  </div>
   158→                </div>
   159→              )}
   160→              {hyp.resultSummary && (
   161→                <div>
   162→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground mb-1.5 font-medium">Result</p>
   163→                  <pre className="text-xs font-terminal text-muted-foreground bg-black/20 rounded-md p-2 overflow-x-auto max-h-32 overflow-y-auto whitespace-pre">
   164→                    {hyp.resultSummary}
   165→                  </pre>
   166→                </div>
   167→              )}
   168→              {hyp.reasoning && (
   169→                <div>
   170→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground mb-1 font-medium">Reasoning</p>
   171→                  <p className="text-sm text-muted-foreground">{hyp.reasoning}</p>
   172→                </div>
   173→              )}
   174→              {hyp.dbMs !== undefined && (
   175→                <div className="flex gap-4 text-xs text-muted-foreground font-terminal">
   176→                  <span>{hyp.dbMs.toFixed(1)}ms</span>
   177→                  {hyp.rowsReturned !== undefined && <span>{hyp.rowsReturned} rows</span>}
   178→                  {hyp.rowsScanned !== undefined && <span>{(hyp.rowsScanned / 1e6).toFixed(0)}M scanned</span>}
   179→                </div>
   180→              )}
   181→            </div>
   182→          </motion.div>
   183→        )}
   184→      </AnimatePresence>
   185→    </motion.div>
   186→  )
   187→}
